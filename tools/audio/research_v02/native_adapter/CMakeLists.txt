cmake_minimum_required(VERSION 3.20)
project(rnnoise_v02_capture_research LANGUAGES C CXX)
if(NOT IS_DIRECTORY "${RNNOISE_SOURCE}/src")
  message(FATAL_ERROR "A verified RNNOISE_SOURCE is required")
endif()
get_filename_component(root "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(capture "${root}/clients/flutter/packages/flutter_webrtc/common/rnnoise")
set(src "${RNNOISE_SOURCE}/src")
add_library(v02 OBJECT
  "${src}/denoise.c" "${src}/rnn.c" "${src}/pitch.c" "${src}/kiss_fft.c"
  "${src}/celt_lpc.c" "${src}/nnet.c" "${src}/nnet_default.c"
  "${src}/parse_lpcnet_weights.c" "${src}/rnnoise_data.c" "${src}/rnnoise_tables.c")
target_include_directories(v02 PRIVATE "${RNNOISE_SOURCE}/include" "${src}" "${CMAKE_CURRENT_SOURCE_DIR}/..")
target_compile_definitions(v02 PRIVATE rnnoise_init=rnnoise_v02_init rnnoise_process_frame=rnnoise_v02_process_frame)
set(processor_sources compat.c
  "${capture}/rnnoise_capture_processor.cc" "${capture}/deepfilter/capture_hook/process.cc"
  "${capture}/microphone_controls/processor.cc" $<TARGET_OBJECTS:v02>)
add_executable(native_v02_adapter_test ${processor_sources} "${capture}/tests/rnnoise_capture_processor_test.cc")
add_executable(native_v02_parity_test ${processor_sources} parity_test.cc)
add_executable(native_v02_16k_test ${processor_sources} parity_16k_test.cc)
target_sources(native_v02_16k_test PRIVATE research_16k.cc)
target_include_directories(native_v02_adapter_test PRIVATE "${capture}")
target_include_directories(native_v02_parity_test PRIVATE "${capture}")
target_include_directories(native_v02_16k_test PRIVATE "${capture}")
target_compile_features(native_v02_adapter_test PRIVATE cxx_std_17)
target_compile_features(native_v02_parity_test PRIVATE cxx_std_17)
target_compile_features(native_v02_16k_test PRIVATE cxx_std_17)
target_compile_definitions(native_v02_16k_test PRIVATE BOOHTA_RNNOISE_V02_RESEARCH)
if(MSVC)
  target_compile_options(v02 PRIVATE /std:c11)
  target_compile_definitions(v02 PRIVATE _USE_MATH_DEFINES _CRT_SECURE_NO_WARNINGS)
  target_compile_options(native_v02_adapter_test PRIVATE /UNDEBUG)
  target_compile_options(native_v02_parity_test PRIVATE /UNDEBUG)
  target_compile_options(native_v02_16k_test PRIVATE /UNDEBUG)
else()
  target_compile_options(native_v02_adapter_test PRIVATE -UNDEBUG)
  target_compile_options(native_v02_parity_test PRIVATE -UNDEBUG)
  target_compile_options(native_v02_16k_test PRIVATE -UNDEBUG)
  target_link_libraries(native_v02_adapter_test PRIVATE m)
  target_link_libraries(native_v02_parity_test PRIVATE m)
  target_link_libraries(native_v02_16k_test PRIVATE m)
endif()
enable_testing()
add_test(NAME native_v02_adapter COMMAND native_v02_adapter_test)
add_test(NAME native_v02_parity COMMAND native_v02_parity_test)
add_test(NAME native_v02_16k COMMAND native_v02_16k_test)
set_tests_properties(native_v02_adapter native_v02_parity native_v02_16k PROPERTIES TIMEOUT 30)
