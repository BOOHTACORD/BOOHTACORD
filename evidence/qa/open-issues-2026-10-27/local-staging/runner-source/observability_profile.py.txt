 for name,pairs in ports.items():
  model['services'][name]['ports']=[{'host_ip':'127.0.0.1','published':str(out),'target':inside,'protocol':'tcp'} for out,inside in pairs]
 model['services']['collector']['networks']={'default':{}}
 model['services']['tempo']['networks']={'default':{}}

 config=WORK/'grafana';(config/'datasources').mkdir(parents=True,exist_ok=True);(config/'dashboards').mkdir(exist_ok=True)
 write(config/'datasources/local.yaml',{'apiVersion':1,'datasources':[

    for child in value.values():bind(child)
   elif isinstance(value,list):
    for child in value:bind(child)
  bind(data);write(rendered/path.name,data);mapping.append({'file':path.name,'source_sha256':hashlib.sha256(original).hexdigest(),'local_sha256':hashlib.sha256((rendered/path.name).read_bytes()).hexdigest()})

  'networks':{'default':{}},'depends_on':{'tempo':{'condition':'service_started'},'prometheus':{'condition':'service_started'}}}
