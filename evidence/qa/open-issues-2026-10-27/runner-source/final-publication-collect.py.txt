"""Import only hashed frozen projections; leave every runtime packet untouched."""
import sys, re
from pathlib import Path
Q = Path(__file__).resolve().parents[2]
sys.path.insert(0, str(Q / '.out/qa-archive-release'))
from common import *
P = Path(__file__).resolve().parent
receipt, proposed = [], []

def copy(src, rel, dst, expected=None):
    source = (src / rel).resolve()
    assert source.is_relative_to(src.resolve())
    raw = source.read_bytes()
    if expected:
        assert sha(raw) == expected['sha256'] and len(raw) == expected['bytes']
    assert Path(rel).suffix in ['.json', '.txt', '.md'] and len(raw) < 300000
    text = raw.decode('utf-8-sig')
    assert not any(n in text for n in needles)
    assert not re.search(r'[A-Za-z]:[/\\]Users[/\\]', text)
    if dst.startswith('publication-bundles/') and Path(rel).suffix == '.md':
        text = re.sub(r'\]\((isolated-[^)]+)\)', r'](../../\1)', text)
    out = O / dst
    assert not out.exists(), dst
    out.parent.mkdir(parents=True, exist_ok=True)
    out.write_text(text.replace('\r\n', '\n'), encoding='utf-8', newline='\n')
    assert source.read_bytes() == raw
    receipt.append({'source_export_relative': rel, 'public_relative_to_root': dst,
                    'prepared_original_SHA256': sha(raw),
                    'final_public_SHA256': sha(out.read_bytes()), 'bytes': out.stat().st_size})
