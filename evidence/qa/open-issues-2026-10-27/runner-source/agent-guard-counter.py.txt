    function=re.search(r'CREATE OR REPLACE FUNCTION protect_support_service_account\(\).*?END \$\$;',source,re.S).group()
    trigger=re.search(r'CREATE TRIGGER support_service_account_guard.*?protect_support_service_account\(\);',source,re.S).group()
    fixed=function.replace(' RETURN NEW;'," IF TG_OP='DELETE' THEN RETURN OLD; END IF;\n RETURN NEW;")
    assert fixed!=function and function.count(' RETURN NEW;')==1
    schema='qa_guard_'+secrets.token_hex(12);before=guard_state()
    statements="""BEGIN; SET LOCAL lock_timeout='2s'; SET LOCAL statement_timeout='5s';

FUNCTION_SOURCE
TRIGGER_SOURCE

""".replace('SCHEMA_NAME',schema).replace('FUNCTION_SOURCE',function).replace('TRIGGER_SOURCE',trigger).replace('COUNTERFACTUAL_SOURCE',fixed)
