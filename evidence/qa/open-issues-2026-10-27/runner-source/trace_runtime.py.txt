def dump(name,value):(OUT/name).write_text(json.dumps(value,indent=2)+'\n',encoding='utf-8',newline='\n')
def command(args):return subprocess.run(args,capture_output=True,text=True,encoding='utf-8',errors='replace')
def control(action,service):
 assert service in ['collector','tempo'];assert action in ['stop','start']
 ids=command(BASE+['ps','-aq',service]);assert ids.returncode==0 and ids.stdout.strip()
 data=json.loads(command(['docker','inspect',ids.stdout.strip()]).stdout)[0]
 assert data['Config']['Labels']['com.docker.compose.project']==PROJECT
 result=command(BASE+[action,service]);assert result.returncode==0,'Owned service control failed'
 return {'service':service,'action':action,'at_utc':datetime.now(timezone.utc).isoformat()}
def probe(url):
 try:
  with urllib.request.build_opener(urllib.request.ProxyHandler({})).open(url,timeout=2) as response:return response.status
 except urllib.error.HTTPError as error:return error.code
 except OSError:return None

if mode=='ready':

 while time.monotonic()<deadline:
  codes={k:probe(u) for k,u in {'tempo':'http://127.0.0.1:13200/ready','collector':'http://127.0.0.1:14318/v1/traces','livekit':'http://127.0.0.1:17880/'}.items()}
  db=command(BASE+['exec','-T','postgres','psql','-U','trace_qa','-d','trace_qa','-Atc','SHOW server_version'])
  if codes['tempo']==200 and codes['collector'] in [400,405] and codes['livekit']==200 and db.returncode==0:break
  time.sleep(1)
 else:raise RuntimeError('Fixture not ready; inspect private container diagnostics')
 images=[]
 for name in ['collector','tempo','postgres','livekit']:
  data=json.loads(command(['docker','inspect',command(BASE+['ps','-q',name]).stdout.strip()]).stdout)[0]
  assert data['Config']['Labels']['com.docker.compose.project']==PROJECT
  image=json.loads(command(['docker','image','inspect',data['Image']]).stdout)[0]
  images.append({'service':name,'image_ref':data['Config']['Image'],'image_id':data['Image'],'repo_digests':image['RepoDigests'],'state':data['State']['Status']})
 result={'at_utc':datetime.now(timezone.utc).isoformat(),'endpoint_codes':codes,'postgres_server_version':db.stdout.strip(),'images':images}
 dump('trace-fixture-ready.json',result);print(json.dumps(result));sys.exit(0)
profiles={
 'relay':(['./internal/observability/ingest_client_traces'],'TestRuntimeRelay|TestRelayHealth|TestRelaySourceFailure'),
 'browser':(['./internal/observability/ingest_client_traces'],'TestBrowserComponents'),

env={k:v for k,v in os.environ.items() if not k.startswith(('TRACE_QA_','OTEL_','VOICE_PLATFORM_TEST_DATABASE_URL','TEST_DATABASE_URL'))}

 'VOICE_PLATFORM_TEST_DATABASE_URL':'postgres://trace_qa:synthetic-qa-only@127.0.0.1:15432/trace_qa?sslmode=disable','TEST_DATABASE_URL':'postgres://trace_qa:synthetic-qa-only@127.0.0.1:15432/trace_qa?sslmode=disable',
 'TRACE_QA_SOURCE_SHA':command(['git','-C',str(ROOT),'rev-parse','HEAD']).stdout.strip(),'CHROMIUM_EXECUTABLE_PATH':r'C:\Program Files\Google\Chrome\Application\chrome.exe'})
if mode=='browser':env['TRACE_QA_BROWSER']='1'
if mode=='native':
 env['TRACE_QA_NATIVE']='1';env['TRACE_QA_FLUTTER_BIN']=json.loads((OUT/'private-toolchain-paths.json').read_text(encoding='utf-8'))['flutter_bin']
if mode=='collector_outage':env['TRACE_QA_EXPECT_COLLECTOR_OUTAGE']='1'
if mode=='tempo_outage':env['TRACE_QA_EXPECT_TEMPO_OUTAGE']='1'
args=['go','test','-json','-count=1','-timeout=8m','-tags','tracing_runtime',*packages,'-run',pattern]
record={'profile':mode,'source_sha':env['TRACE_QA_SOURCE_SHA'],'command':subprocess.list2cmdline(args),'cwd':'backend','at_utc':datetime.now(timezone.utc).isoformat(),'scope':'isolated components, not packaged physical clients','events':[]}

  restore='collector' if mode=='collector_outage' else 'tempo';history.append(control('stop',restore))

sys.exit(record.get('exit',1))
