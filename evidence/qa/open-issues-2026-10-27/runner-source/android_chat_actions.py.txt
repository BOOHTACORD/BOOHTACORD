"""Operate only the selected owned-peer DM, never print content or identity."""
import hashlib
import json
import re
import subprocess
import sys
import xml.etree.ElementTree as ET
from datetime import datetime, timezone
from pathlib import Path

ROOT=Path(__file__).parent
ADB=r'C:\Users\egkur\AppData\Local\Android\Sdk\platform-tools\adb.exe'
peer=json.loads((ROOT/'private-fixture.json').read_text(encoding='utf-8'))['b']['login']


def call(*args):
    return subprocess.check_output([ADB,*args],timeout=30).decode('utf-8','replace')


def nodes():
    raw=call('exec-out','uiautomator','dump','/dev/tty')
    return list(ET.fromstring(raw[raw.find('<hierarchy'):raw.find('</hierarchy>')+12]).iter('node'))


def label(n):
    return n.get('content-desc') or n.get('text') or ''


def bounds(n):
    return list(map(int,re.findall(r'\d+',n.get('bounds'))))


def tap(matches,long=False):
    if len(matches)!=1:raise RuntimeError('Expected exactly one bounded control')
    b=bounds(matches[0]);x,y=str((b[0]+b[2])//2),str((b[1]+b[3])//2)
    if b[3]-b[1]>(500 if matches[0].get('class')=='android.widget.EditText' else 250):raise RuntimeError('Refusing aggregate container')
    call('shell','input',*(['swipe',x,y,x,y,'650'] if long else ['tap',x,y]))


rows=nodes();action=sys.argv[1]
if not any(peer in label(n) and bounds(n)[1]<600 for n in rows):
    if action not in ('cancel','delete','edit','save','append','back'):raise RuntimeError('Owned peer must be visible')
if action=='type':
    fields=[n for n in rows if n.get('class')=='android.widget.EditText']
    if len(fields)!=1 or fields[0].get('text'):raise RuntimeError('Expected empty selected composer')
    if call('shell','settings','get','system','font_scale').strip()!='1.0':raise RuntimeError('Screenshot geometry requires original1x')
    for x,y in [(631,1976),(443,1695),(352,1976),(631,1976)]:call('shell','input','tap',str(x),str(y))
elif action=='back':call('shell','input','keyevent','KEYCODE_BACK')
elif action=='send':tap([n for n in rows if n.get('clickable')=='true' and label(n).startswith('Отправить') and n.get('package')=='ru.boohtacord.app'])
elif action=='longpress':tap([n for n in rows if label(n) in ('тест','тестт') and n.get('class')!='android.widget.EditText'],long=True)
elif action=='menu':tap([n for n in rows if 'Действия с сообщением' in label(n) and n.get('clickable')=='true'])
elif action=='react':tap([n for n in rows if '👍' in label(n) and n.get('clickable')=='true' and n.get('package')=='ru.boohtacord.app'])
elif action in ('cancel','delete','edit','save'):
    text={'cancel':'Отмена','delete':'Удалить','edit':'Изменить','save':'Сохранить'}[action]
    tap([n for n in rows if label(n)==text and n.get('clickable')=='true' and n.get('package')=='ru.boohtacord.app'])
elif action=='append':
    fields=[n for n in rows if n.get('class')=='android.widget.EditText' and n.get('text')=='тест']
    if len(fields)!=1:raise RuntimeError('Expected selected owned test editor')
    tap(fields);nodes();call('shell','input','keyevent','KEYCODE_MOVE_END');call('shell','input','tap','631','1976')
elif action!='inspect':raise ValueError('Unknown bounded action')
rows=nodes();texts=[label(n) for n in rows]
services=call('shell','dumpsys','activity','services','ru.boohtacord.app')
report={'action':action,'at_utc':datetime.now(timezone.utc).isoformat(),
        'runner_sha256':hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
        'voice_connected':any('Голос подключён' in s for s in texts),
        'microphone_fgs_active':'types=0x00000080' in services,
        'composer_fields':[{'length':len(n.get('text') or ''),'matches_four_char_fixture':n.get('text')=='тест','matches_five_char_fixture':n.get('text')=='тестт','bounds':bounds(n)} for n in rows if n.get('class')=='android.widget.EditText'],
        'visible_fixture_nodes':sum(s in ('тест','тестт') for s in texts),
        'menu_controls':[{'control':label(n),'bounds':bounds(n)} for n in rows if label(n) in ['Ответить','Изменить','Удалить','Отмена','Сохранить','Реакции','Да'] and n.get('clickable')=='true'],
        'available_actions':[{'control':k,'bounds':bounds(n),'clickable':n.get('clickable'),'class':n.get('class')} for n in rows for k in ['Действия с сообщением','Ответить','Изменить','Удалить','Копировать','Выделить все','Реакции'] if k in label(n)],
        'reaction_controls':[{'emoji':k,'bounds':bounds(n),'clickable':n.get('clickable')} for n in rows for k in ['👍','❤️','😂','🎉','👀','✅'] if k in label(n) and n.get('clickable')=='true']}
if action=='type':report['keyboard_source_sha256']=hashlib.sha256((ROOT/'native-ime-settled.png').read_bytes()).hexdigest()
(ROOT/('native-chat-'+action+'.json')).write_text(json.dumps(report,ensure_ascii=False,indent=2)+'\n',encoding='utf-8')
print(json.dumps(report,ensure_ascii=True))
