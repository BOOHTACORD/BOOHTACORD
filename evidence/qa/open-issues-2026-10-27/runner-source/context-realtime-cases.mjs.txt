 const row=main(page).locator('[data-message-id="'+cfg.ids[189]+'"]');await row.scrollIntoViewIfNeeded();const before=hintCount(189);const changed='Synthetic realtime main control '+cfg.owner

export async function lifecycle(page,i,replyIndex,mode){
 await fresh(page);const trigger=await open(page,i,replyIndex);await expect(target(page)).toContainText(bodyOriginal(i));const before=await rowState(page,i),hints=hintCount(i)
 const path='/channels/'+cfg.main.channel+'/messages/'+cfg.ids[i],changed='Synthetic changed row '+i+' '+cfg.owner
 const original=await actor('/channels/'+cfg.main.channel+'/messages?at='+cfg.ids[i]+'&limit=20');assert.equal(original.status,200);assert.equal(original.body.messages[0].body,bodyOriginal(i));const initialRevision=original.body.messages[0].revision
 const r=await actor(path,mode==='edit'?'PATCH':'DELETE',mode==='edit'?{body:changed,expected_revision:initialRevision}:undefined);assert.equal(r.status,mode==='edit'?200:204)
 const freshPage=await api(page,'/channels/'+cfg.main.channel+'/messages?at='+cfg.ids[i]+'&limit=20');assert.equal(freshPage.status,200);assert.equal(freshPage.body.messages[0].id,cfg.ids[i]);const server=freshPage.body.messages[0]
 if(mode==='edit'){assert.equal(server.body,changed);assert.equal(server.revision,initialRevision+1)}else{assert.equal(server.deleted,true);assert.equal(server.body,'');assert.deepEqual(server.attachments,[])}
 await expect.poll(()=>hintCount(i)).toBeGreaterThan(hints);const observed=await observe(page,i);const final=observed.samples.at(-1),actual=mode==='edit'?final.contains_changed&&!final.contains_original:final.tombstone&&!final.contains_original

 const result={status:actual?'PASS':'FAIL_LIVE_CONTEXT_STALE',mode,target_index:i,reply_index:replyIndex,before,actor_HTTP:r.status,server_at_HTTP:200,initial_server_revision:initialRevision,server_revision:server.revision,server_deleted:server.deleted===true,server_attachments_empty:server.attachments.length===0,matched_target_hint_delta:hintCount(i)-hints,observed,expected_live_update:actual,scope:'Owned TEXT authorized peer, actual protected API actor mutation and real receiver WebSocket. }
