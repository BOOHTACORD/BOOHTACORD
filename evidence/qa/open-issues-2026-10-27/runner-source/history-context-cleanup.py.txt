"""Remove only attested own scopes and revoke every session created by this packet."""
import csv,hashlib,io,ipaddress,json,subprocess
from datetime import datetime,timezone
from common import P,Q,PROJECT,read,write,sql,valid,request,docker
assert not (P/'environment-after.json').exists()
assert read(P/'semantic-result.json')['status'].startswith('PASS')
c=read(P/'private/config.private.json');before=read(P/'environment-before.json');category=valid(c['category']);channels=[valid(d['channel']) for d in c['channels'].values()]
assert len(channels)==len(set(channels))==3 and len(c['owner_token'])==24 and all(x in '0123456789abcdef' for x in c['owner_token'])
assert sql("SELECT name FROM categories WHERE id='"+category+"';").strip()=='qa_context_'+c['owner_token']
assert set(sql("SELECT id FROM channels WHERE category_id='"+category+"';").splitlines())==set(channels)
r={'at_utc':datetime.now(timezone.utc).isoformat(),'status':'RUNNING','production_touched':False,'dataset_preservation':[]}
for alias,d in c['channels'].items():
    command="COPY (SELECT id,channel_id,author_id,client_message_id,body,created_at,reply_to_id FROM messages WHERE channel_id='"+valid(d['channel'])+"' ORDER BY created_at,id) TO STDOUT WITH CSV;"
    raw=docker('exec','-i',PROJECT+'-postgres-1','psql','-U','boohtacord_qa','-d','boohtacord_qa','-qAt','-v','ON_ERROR_STOP=1',input=command.encode()).decode();data=list(csv.reader(io.StringIO(raw)));assert [x[0] for x in data]==d['ids']
    for row in data:row[5]=datetime.fromisoformat(row[5]).astimezone(timezone.utc).isoformat()
    out=io.StringIO(newline='');csv.writer(out,lineterminator='\n').writerows(data);digest=hashlib.sha256(out.getvalue().encode()).hexdigest();assert digest==d['seed_CSV_SHA256']
    r['dataset_preservation'].append({'alias':alias,'count':len(data),'seed_CSV_SHA256':digest,'server_data_matches_seed':True})
scopes=','.join("'"+x+"'" for x in channels)
assert int(sql('SELECT count(*) FROM messages WHERE channel_id IN ('+scopes+') AND (revision<>1 OR deleted_at IS NOT NULL);').strip())==0
assert int(sql('SELECT count(*) FROM attachments WHERE channel_id IN ('+scopes+');').strip())==0
assert int(sql('SELECT count(*) FROM guild_settings WHERE welcome_channel_id IN ('+scopes+');').strip())==0
sql('BEGIN;\nSET LOCAL statement_timeout=\'90s\';\nDELETE FROM channel_read_cursors WHERE channel_id IN ('+scopes+');\nDELETE FROM messages WHERE channel_id IN ('+scopes+');\nDELETE FROM channels WHERE category_id=\''+category+'\';\nDELETE FROM categories WHERE id=\''+category+"' AND name='qa_context_"+c['owner_token']+"';\nCOMMIT;\n")
assert sql('SELECT count(*) FROM messages WHERE channel_id IN ('+scopes+');\nSELECT count(*) FROM channels WHERE category_id=\''+category+'\';\nSELECT count(*) FROM categories WHERE id=\''+category+'\';').splitlines()==['0','0','0']
status,_,t=request('/api/v1/channels',cookie=c['cookie']);assert status==200 and not {x.get('id') for cat in t['categories'] for x in cat['channels']} & set(channels) and all(x['id']!=category for x in t['categories']);r['actual_API_topology_own_scopes_absent']=True
assert len(c['sessions'])==len(set(c['sessions']));revoked=0;already=0
for cookie in c['sessions']:
    status,_,session=request('/api/v1/auth/session',cookie=cookie);assert status==200
    if session['authenticated']:
        status,_,_=request('/api/v1/auth/logout','POST',{},cookie);assert status==204;revoked+=1
    else:already+=1
    status,_,session=request('/api/v1/auth/session',cookie=cookie);assert status==200 and session['authenticated'] is False
values=sql('SELECT (SELECT count(*) FROM users),(SELECT count(*) FROM sessions),(SELECT count(*) FROM sessions WHERE revoked_at IS NULL),(SELECT count(*) FROM attachments),(SELECT count(*) FROM messages);\nSHOW server_version;').splitlines();counts=list(map(int,values[0].split('|')));expected=list(before['database_counts']);expected[1]+=len(c['sessions']);assert counts==expected
r.update(database_counts_before=before['database_counts'],database_counts_after=counts,database_count_order=['users','all_sessions','unrevoked_rows_not_validity','attachments','messages'],PostgreSQL_version=values[1],own_fixture_messages_removed=1320,own_channels_removed=3,own_category_removed=1,own_revoked_session_audit_rows_retained=len(c['sessions']),own_sessions_already_revoked_by_runner=already,own_sessions_revoked_at_cleanup=revoked,all_own_sessions_live_unauthenticated_readback=True,services=[])
for name in ['api','web','proxy','postgres','livekit','collector','tempo','prometheus','grafana']:
    old=read(P/'private'/(name+'-inspect-before.private.json'));v=json.loads(docker('inspect',old['Id']))[0]
    assert v['State']['Running'] and v['RestartCount']==old['RestartCount']==0 and v['Image']==old['Image'] and v['Config']==old['Config'] and v['HostConfig']==old['HostConfig'] and v['State']['StartedAt']==old['State']['StartedAt']
    for bindings in v['NetworkSettings']['Ports'].values():
        for b in bindings or []:assert ipaddress.ip_address(b['HostIp']).is_loopback
    r['services'].append({'service':name,'same_ID_image_Config_HostConfig_StartTime':True,'restart_count':0,'published_ports_loopback':True})
status,_,_=request('/api/v1/health');assert status==200;r['API_health_http']=status
assert not subprocess.check_output(['git','-c','core.autocrlf=false','status','--porcelain'],cwd=Q,text=True).strip();r['source_worktree_clean']=True
r['status']='PASS_1320_ROWS_THREE_SCOPES_ALL_OWN_SESSIONS_REMOVED_NINE_SERVICES_UNCHANGED';r['finished_at_utc']=datetime.now(timezone.utc).isoformat();write(P/'environment-after.json',r)
print(json.dumps({k:v for k,v in r.items() if k not in ['dataset_preservation','services']}))
