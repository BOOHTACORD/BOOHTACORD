original = {k: r['requirement'] for k, r in rows.items()}

def add(c, ids):
    assert not any(old['id'] == c['id'] for old in cases['cases'])
    assert set(ids) <= {'I240.C17', 'I287.C8', 'I62.C1'}
    assert c['issues'] == sorted({int(r.split('.')[0][1:]) for r in ids})
    for key in ['protects', 'command', 'platform', 'mandatory_when', 'result', 'limits']:
        assert isinstance(c.get(key), str) and c[key].strip()
    path, _, pointer = c['result'].partition('#')
    value = read(O / path)
    if pointer:
        assert pointer.startswith('/')
        for part in pointer[1:].split('/'):
            part = part.replace('~1', '/').replace('~0', '~')
            value = value[int(part)] if isinstance(value, list) else value[part]
    cases['cases'].append(c)
    for rid in ids:
        r = rows[rid]
        r['execution_cases'].append(c['id'])
        r['evidence'].append(c['result'])
        r.setdefault('witness_limits', []).append(c['id'] + ': ' + c['limits'])
        r['execution'] += '; ' + c['command']
        if 'execution_plan' in r: r['execution_plan']['existing_scoped_witnesses'] = list(r['execution_cases'])
    for issue in m['issues']:
        if issue['number'] in c['issues']:
            issue['execution_cases'].append(c['id'])
            issue['evidence'].append(c['result'])

for c in read(P / 'proposed-cases.json')['cases']:
