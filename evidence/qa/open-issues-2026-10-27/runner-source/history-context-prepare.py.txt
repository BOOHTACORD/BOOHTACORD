"""Create uniquely owned long-history/context fixtures using the actual protected API."""
import csv,hashlib,io,secrets,uuid
from datetime import datetime,timedelta,timezone
from http.cookies import SimpleCookie
from common import P,SOURCE,ORIGIN,read,write,credentials,request,sql,valid
path=P/'private/config.private.json'
assert not path.exists(),'Never replace an owned fixture receipt'
assert read(P/'environment-before.json')['source_revision']==SOURCE
c=credentials();cfg={'owner_token':secrets.token_hex(12),'origin':ORIGIN,'source_revision':SOURCE,'channels':{},'sessions':[]}
write(path,cfg)
status,headers,_=request('/api/v1/auth/login','POST',{'login':c['admin_login'],'password':c['admin_password']});assert status==204
cookies=SimpleCookie()
for k,v in headers:
    if k.lower()=='set-cookie':cookies.load(v)
assert len(cookies)==1;cfg['cookie']='; '.join(k+'='+v.value for k,v in cookies.items());cfg['sessions'].append(cfg['cookie']);write(path,cfg)
status,_,s=request('/api/v1/auth/session',cookie=cfg['cookie']);assert status==200 and s['authenticated'] and s['role']=='ADMINISTRATOR'
cfg['account']=valid(s['account_id']);write(path,cfg)
cfg['peer_account']=valid(sql("SELECT id FROM users WHERE login='"+c['member_a_login'].replace("'","''")+"';").strip());write(path,cfg)
status,_,cat=request('/api/v1/admin/categories','POST',{'name':'qa_context_'+cfg['owner_token']},cfg['cookie']);assert status==201
cfg['category']=valid(cat['id']);write(path,cfg)
epoch=datetime(2026,1,1,tzinfo=timezone.utc)
for alias,count in [('long',1200),('control',0),('unread',120)]:
    name='qa_'+alias+'_'+cfg['owner_token'];status,_,body=request('/api/v1/admin/categories/'+cfg['category']+'/channels','POST',{'name':name,'kind':'TEXT'},cfg['cookie']);assert status==201
    d={'channel':valid(body['id']),'name':name,'ids':[str(uuid.uuid4()) for _ in range(count)],'count':count}
    cfg['channels'][alias]=d;write(path,cfg)
    out=io.StringIO(newline='');writer=csv.writer(out,lineterminator='\n')
    for i,identifier in enumerate(d['ids']):
        text='Synthetic context row '+str(i)+'\n'+'Synthetic height line\n'*(i%3)
        if alias=='long' and i==300:text+='qa_context_search_'+cfg['owner_token']
        writer.writerow([identifier,d['channel'],cfg['peer_account'] if alias=='unread' else cfg['account'],str(uuid.uuid4()),text,(epoch+timedelta(seconds=i)).isoformat(),d['ids'][300] if alias=='long' and i==1199 else None])
    raw=out.getvalue();d['seed_CSV_SHA256']=hashlib.sha256(raw.encode()).hexdigest();write(path,cfg)
    (P/'private'/(alias+'.private.csv')).write_text(raw,encoding='utf-8',newline='\n')
    if count:sql('BEGIN;\nCOPY messages(id,channel_id,author_id,client_message_id,body,created_at,reply_to_id) FROM STDIN CSV;\n'+raw+'\\.\nCOMMIT;\n')
    assert int(sql("SELECT count(*) FROM messages WHERE channel_id='"+d['channel']+"';").strip())==count
    if count:
        status,_,page=request('/api/v1/channels/'+d['channel']+'/messages',cookie=cfg['cookie']);assert status==200 and len(page['messages'])==50 and page['messages'][0]['id']==d['ids'][-1]
unread=cfg['channels']['unread'];status,_,cursor=request('/api/v1/channels/'+unread['channel']+'/read-cursor','PUT',{'message_id':unread['ids'][49]},cfg['cookie']);assert status==200 and cursor['message_id']==unread['ids'][49]
status,_,topology=request('/api/v1/channels',cookie=cfg['cookie']);assert status==200
for alias,d in cfg['channels'].items():
    row=next(x for cat in topology['categories'] for x in cat['channels'] if x['id']==d['channel'])
    if alias=='unread':
        (P/'private/unread-topology-before.private.json').write_text(__import__('json').dumps(row))
        assert row['first_unread_message_id']==d['ids'][50] and row['unread_count']==70
write(P/'fixture-ready.json',{'at_utc':datetime.now(timezone.utc).isoformat(),'status':'PASS_OWNED_THREE_CHANNELS_1320_ROWS_REAL_PROTECTED_API','source_revision':SOURCE,'datasets':[{k:v for k,v in d.items() if k in ['count','seed_CSV_SHA256']}|{'alias':a} for a,d in cfg['channels'].items()],'search_and_reply_target_index':300,'target_outside_latest_50':True,'unread_precondition':{'last_read_index':49,'first_unread_index':50,'unread_count':70},'production_touched':False,'IDs_CSV_bodies_cookies_private':True})
print('PASS_OWNED_THREE_CHANNELS_1320_ROWS_REAL_PROTECTED_API')
