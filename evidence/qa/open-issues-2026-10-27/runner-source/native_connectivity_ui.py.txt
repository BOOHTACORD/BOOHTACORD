"""Only the selected own DM fixture may be sent/deleted; no contents or identities emitted."""
import json,re,sys,xml.etree.ElementTree as ET
from datetime import datetime,timezone
from android_permission_ui import APP,ROOT,call,nodes,label,tap_named
cfg=json.loads((ROOT/'private-connectivity-fixture.json').read_text(encoding='utf-8'));text=cfg['text'];peer=cfg['peer']
def box(n):return list(map(int,re.findall(r'\d+',n.get('bounds'))))
def tap(matches):
 assert len(matches)==1,'Expected one observed fixture control';b=box(matches[0]);assert 0<b[3]-b[1]<500
 call('shell','input','tap',str((b[0]+b[2])//2),str((b[1]+b[3])//2))
def own(rows):assert any(peer in label(n) and box(n)[1]<700 for n in rows),'Owned DM header required'
action=sys.argv[1];rows=nodes()
if action=='delete':assert json.loads((ROOT/'connectivity-ui-delete-menu.json').read_text(encoding='utf-8'))['fixture_scoped_menu_activated']
else:own(rows)
report={'at_utc':datetime.now(timezone.utc).isoformat(),'action':action,'body_or_identity_logged':False}
if action=='send':
 fields=[n for n in rows if n.get('class')=='android.widget.EditText'];assert len(fields)==1 and not fields[0].get('text')
 tap(fields);call('shell','input','text',text);rows=nodes();own(rows)
 assert any(n.get('class')=='android.widget.EditText' and n.get('text')==text for n in rows)
 send=[n for n in rows if n.get('package')==APP and n.get('clickable')=='true' and label(n).startswith('Отправить')];assert len(send)==1
 leave=[n for n in rows if label(n)=='Выйти из голосового канала'];assert len(leave)==1
 a,b=box(send[0]),box(leave[0]);assert min(a[3],b[3])<=max(a[1],b[1]) or min(a[2],b[2])<=max(a[0],b[0]),'Refuse known Send/Leave overlap'
 report['send_bounds']=a;report['leave_bounds']=b;tap(send);report['owned_fixture_send_control_activated']=True
elif action=='delete-menu':
 if 'mInputShown=true' in call('shell','dumpsys','input_method'):call('shell','input','keyevent','KEYCODE_BACK')
 raw=call('exec-out','uiautomator','dump','/dev/tty');tree=ET.fromstring(raw[raw.find('<hierarchy'):raw.find('</hierarchy>')+12]);parents={c:p for p in tree.iter() for c in p}
 matches=[n for n in tree.iter('node') if text in label(n) and n.get('class')!='android.widget.EditText' and not any(text in label(c) for c in n.iter('node') if c is not n)]
 assert len(matches)==1,'Expected one owned live fixture message'
 ancestor=matches[0];controls=[]
 while ancestor in parents:
  ancestor=parents[ancestor];controls=[n for n in ancestor.iter('node') if n.get('package')==APP and n.get('clickable')=='true' and 'Действия с сообщением' in label(n)]
  if controls:break
 assert len(controls)==1 and box(ancestor)[3]-box(ancestor)[1]<1200
 tap(controls);report['fixture_scoped_menu_activated']=True
elif action=='delete':tap_named('Удалить');report['observed_delete_control_activated']=True
elif action=='clear':
 fields=[n for n in rows if n.get('class')=='android.widget.EditText'];assert len(fields)==1
 value=fields[0].get('text') or '';assert not value or value==text,'Refuse unknown draft'
 if value:
  tap(fields);call('shell','input','keyevent','KEYCODE_MOVE_END',*(['67']*len(value)))
 if 'mInputShown=true' in call('shell','dumpsys','input_method'):call('shell','input','keyevent','KEYCODE_BACK')
 report['own_draft_clear_dispatched']=True
elif action=='inspect':
 fields=[n for n in rows if n.get('class')=='android.widget.EditText'];report['composer_empty']=len(fields)==1 and not fields[0].get('text')
 report['fixture_display_nodes']=sum(text in label(n) for n in rows if n.get('class')!='android.widget.EditText')
else:raise ValueError('Unknown fixture action')
(ROOT/f'connectivity-ui-{action}.json').write_text(json.dumps(report,indent=2)+'\n',encoding='utf-8',newline='\n');print(json.dumps(report))
