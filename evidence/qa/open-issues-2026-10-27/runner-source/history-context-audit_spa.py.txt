category=valid(c['category']);channel=valid(c['channel']);message=valid(c['message']);assert len(c['owner'])==32 and all(v in '0123456789abcdef' for v in c['owner'])

for cookie in c['sessions']:
    status,_,s=request('/api/v1/auth/session',cookie=cookie);assert status==200
    if s['authenticated']:status,_,_=request('/api/v1/auth/logout','POST',{},cookie);assert status==204;revoked+=1

logs=[]

    old=read(P/'private'/(name+'-inspect-before.private.json'));v=json.loads(docker('inspect',old['Id']))[0]

    for bindings in v['NetworkSettings']['Ports'].values():
        for b in bindings or []:assert ipaddress.ip_address(b['HostIp']).is_loopback
    result['services'].append({'service':name,'same_ID_image_Config_HostConfig_StartTime':True,'restart_count':0,'published_ports_loopback':True})
