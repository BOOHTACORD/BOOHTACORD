"""One observed Android UI/input action per call; never emit names or message contents."""
import json,re,sys
from datetime import datetime,timezone
from pathlib import Path
from android_permission_ui import APP,ROOT,call,nodes,label,state,tap_named,permission
ALLOWED=['Открыть навигацию','Закрыть навигацию','Настройки аккаунта','Настройки аудио','Активация микрофона','По голосу','По нажатию','Удерживайте, чтобы говорить','Включить микрофон','Выключить микрофон','Выйти из голосового канала']
def snapshot():
 rows=nodes();data=state();data['controls']=[{k:n.get(k) for k in ['bounds','clickable','enabled','checked','selected']}|{'label':label(n)} for n in rows if label(n) in ALLOWED]
 return data
def non_ui_snapshot():
 services=call('shell','dumpsys','activity','services',APP)
 return {'at_utc':datetime.now(timezone.utc).isoformat(),'permission':permission(),'foreground_types':re.findall(r'isForeground=true[^\n]*types=(0x[0-9a-f]+)',services),
  'ui_label_state_not_queried':True,'voice_connection_or_mute_not_inferred_from_FGS':True}
def point():
 rows=[n for n in nodes() if n.get('package')==APP and label(n)=='Удерживайте, чтобы говорить' and n.get('enabled')=='true']
 assert len(rows)==1,'Expected one actual PTT semantic control'
 box=list(map(int,re.findall(r'\d+',rows[0].get('bounds'))));assert box[2]>box[0] and box[3]>box[1]
 return [(box[0]+box[2])//2,(box[1]+box[3])//2]
action=sys.argv[1]
if action in ['account','audio','ptt','vad','nav']:
 tap_named({'account':'Настройки аккаунта','audio':'Настройки аудио','ptt':'По нажатию','vad':'По голосу','nav':'Открыть навигацию'}[action])
elif action=='ptt-observed':
 import hashlib
 from PIL import Image
 file=ROOT/'private-ptt-audio-scrolled.png';assert Image.open(file).size==(1080,2340)
 (ROOT/'ptt-observed-selection-proof.json').write_text(json.dumps({'screenshot_sha256':hashlib.sha256(file.read_bytes()).hexdigest(),'point':[750,1295],'observed_option':'По нажатию','original_selected_option':'По голосу'}),encoding='utf-8')
 call('shell','input','tap','750','1295')
elif action in ['down','up','up-outside']:
 file=ROOT/'private-ptt-point.json'
 if action=='down':
  coordinates=point();file.write_text(json.dumps(coordinates),encoding='utf-8')
 else:coordinates=json.loads(file.read_text(encoding='utf-8'))
 if action=='up-outside':coordinates=[coordinates[0],max(110,coordinates[1]-400)]
 call('shell','input','touchscreen','motionevent','DOWN' if action=='down' else 'UP',*map(str,coordinates))
elif action=='home':call('shell','input','keyevent','KEYCODE_HOME')
elif action=='back':call('shell','input','keyevent','KEYCODE_BACK')
elif action=='resume':call('shell','am','start','-n',APP+'/.MainActivity')
elif action=='help':
 text=call('shell','input');print(json.dumps({'motion_event_help':[line.strip() for line in text.splitlines() if 'motionevent' in line or 'DOWN' in line]}));sys.exit(0)
elif action!='inspect':raise ValueError('Unknown action')
if action in ['down','up','up-outside','home']:data=non_ui_snapshot()
else:
 try:data=snapshot()
 except Exception as error:
  data=non_ui_snapshot();data['ui_observation_error_type']=type(error).__name__
data['action']=action
(ROOT/f'ptt-ui-{action}.json').write_text(json.dumps(data,ensure_ascii=False,indent=2)+'\n',encoding='utf-8',newline='\n')
print(json.dumps(data,ensure_ascii=False))
