import subprocess,time
from common import P,Q,WORK,CONTEXT,PROJECT,read,write,fixture,quote,sql,request,state,services,frozen,sha,now
def cleanup():
    c=fixture();before=read(P/'before.json');target=c['accounts']['a'];owner=c['owner'];assert len(owner)==24 and all(v in '0123456789abcdef' for v in owner)
    admin=c['sessions'].get('admin',{}).get('cookie')
    if admin:
        code,_,lift=request('/api/v1/admin/accounts/'+target+'/voice-timeout','DELETE',cookie=admin);assert code==200 and not lift['active'];write(P/'normal-lift.json',{'utc':now(),'HTTP':code,'state':lift,'scope':'Normal lift; owned work cleanup is separate'})
    assert sql('SELECT count(*) FROM voice_timeouts WHERE user_id='+quote(target)+';').strip()=='0'
    if c.get('lease'):
        assert request('/api/v1/voice/leases/'+c['lease'],'DELETE',cookie=c['sessions']['a']['cookie'])[0]==204
        for _ in range(12):
            busy=int(sql('SELECT count(*) FROM voice_sfu_revocations WHERE lease_id='+quote(c['lease'])+' AND claim_token IS NOT NULL;').strip())
            if not busy:break
            time.sleep(.15)
        assert not busy,'Work still claimed; stop cleanup rather than racing worker'
        assert sql('SELECT count(*) FROM voice_leases WHERE id='+quote(c['lease'])+' AND user_id='+quote(target)+' AND channel_id='+quote(c['channel'])+' AND revoked_at IS NOT NULL;').strip()=='1'
        sql('BEGIN; DELETE FROM voice_sfu_revocations WHERE lease_id='+quote(c['lease'])+'; DELETE FROM voice_leases WHERE id='+quote(c['lease'])+' AND user_id='+quote(target)+' AND revoked_at IS NOT NULL; COMMIT;')
    if c.get('channel'):
        assert sql('SELECT name FROM categories WHERE id='+quote(c['category'])+';').strip()=='qa_dispatch_'+owner and sql('SELECT id FROM channels WHERE category_id='+quote(c['category'])+';').strip()==c['channel']
        assert sql('SELECT count(*) FROM messages WHERE channel_id='+quote(c['channel'])+';').strip()=='0'
        sql('BEGIN; DELETE FROM channels WHERE id='+quote(c['channel'])+'; DELETE FROM categories WHERE id='+quote(c['category'])+' AND name='+quote('qa_dispatch_'+owner)+'; COMMIT;')
    revoked=[]
    for alias,s in c['sessions'].items():
        code,_,session=request('/api/v1/auth/session',cookie=s['cookie']);assert code==200
        if session['authenticated']:assert request('/api/v1/auth/logout','POST',{},s['cookie'])[0]==204
        assert request('/api/v1/channels',cookie=s['cookie'])[0]==401
        assert sql('SELECT count(*) FROM sessions WHERE public_id='+quote(s['id'])+' AND user_id='+quote(s['account'])+' AND revoked_at IS NOT NULL;').strip()=='1'
        sql('DELETE FROM sessions WHERE public_id='+quote(s['id'])+' AND user_id='+quote(s['account'])+' AND revoked_at IS NOT NULL;');revoked.append(alias)
    assert state()==before['database'];old=read(P/'private/services-before.json');current=services();assert all(v['State']['Running'] and all(v[k]==old[n][k] for k in ['Id','Image','Config','HostConfig','RestartCount']) and v['State']['StartedAt']==old[n]['State']['StartedAt'] for n,v in current.items())
    assert request('/api/v1/health')[0]==200 and frozen()==before['frozen_files']
    assert all(sha((Q/f).read_bytes())==h for f,h in before['source_hashes'].items()) and all(sha((P/f).read_bytes())==h for f,h in before['helper_hashes'].items())
    log=subprocess.run(['docker','--context',CONTEXT,'logs','--since',before['utc'],PROJECT+'-api-1'],capture_output=True,timeout=15);raw=log.stdout+log.stderr;(P/'private/api.log').write_bytes(raw)
    cred=read(WORK/'credentials.private.json');needles=[owner,*c['accounts'].values()]+[c[k] for k in ['category','channel','lease'] if k in c]+[cred[k] for k in ['admin_login','admin_password','member_a_login','member_a_password','member_b_login','member_b_password','database_password']]
    for v in c['sessions'].values():needles += [v['cookie'],v['cookie'].split('=',1)[1],v['id']]
    assert not any(v.encode() in raw for v in needles if v)
    write(P/'cleanup.json',{'utc':now(),'status':'PASS','target_timeout_none_by_normal_lift':True,'one_owned_lease_and_intent_removed_after_capture':True,'own_sessions_revoked_live401_removed':revoked,'selected_table_baselines_restored':len(before['database']),'nine_services_unchanged':list(current),'source_helpers_frozen_previous_packets_unchanged':True,'health_HTTP':200,'topology_revision_before':before['topology_revision'],'topology_revision_after':int(sql('SELECT revision FROM channel_topology_state WHERE singleton;').strip()),'audit_metadata_delta':int(sql('SELECT count(*) FROM audit_events;').strip())-before['audit_count'],'API_log_SHA256':sha(raw),'known_privatevalue_log_matches':0,'limit':'Expected monotonic topology/owned audit retained; selected baselines only, no entireDB equality. No participant/SFU room was created.'})
