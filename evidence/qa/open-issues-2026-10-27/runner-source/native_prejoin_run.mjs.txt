import assert from 'node:assert/strict'
import {createRequire} from 'node:module'
import {execFileSync} from 'node:child_process'
import {readFile,writeFile} from 'node:fs/promises'
import {fileURLToPath} from 'node:url'
const {request}=createRequire('C:/Users/egkur/.codex/worktrees/test-sql-native-gates/BOOHTACORD/clients/web/package.json')('@playwright/test')
const root=new URL('./',import.meta.url),origin='https://v.bootybay.ru'

const context=await request.newContext({storageState:JSON.parse(await readFile(new URL('private-session-b.json',root),'utf8'))})

try{
 const directs=await read('/direct-messages');const candidates=directs.direct_messages.filter(d=>![input.a.id,input.c.id].includes(d.other_participant_id));assert.equal(candidates.length,1);phone=candidates[0].other_participant_id
 const before=await read('/voice/participants');const occupied=before.channels.filter(c=>c.participants.length>0);assert.equal(occupied.length,1);assert.equal(occupied[0].participants.length,3);channelId=occupied[0].channel_id
 ownName=occupied[0].participants.find(p=>p.account_id===phone)?.display_name;assert.ok(ownName)
 report.before={participants:3,muted:occupied[0].participants.filter(p=>p.microphone_muted).length}
 started=true;report.selected=run('select-denied')
 const after=(await read('/voice/participants')).channels.find(c=>c.channel_id===channelId);assert.equal(after.participants.length,2);assert.ok(after.participants.every(p=>p.account_id!==phone))
 names=after.participants.map(p=>p.display_name);assert.equal(new Set(names).size,2)
 report.prejoin=run('inspect',{QA_PREJOIN_NAMES:JSON.stringify(names),QA_PREJOIN_SELF:ownName})
 report.after={participants:2,muted:after.participants.filter(p=>p.microphone_muted).length}
 const s=report.prejoin;assert.equal(s.permission.granted,false);assert.equal(s.voice_connected,false);assert.equal(s.microphone_fgs,false);assert.equal(s.system_permission_prompt,false);assert.equal(s.expected_member_names_matched,2);assert.equal(s.self_member_visible,false);assert.ok(s.manual_join_controls.length>=2);assert.equal(s.empty_roster_visible,false)
 report.status='PASS_SCOPED'
}catch(e){report.status='FAIL';report.error_type=e.name}
finally{
 if(started){try{report.restored=run('restore');original=JSON.parse(await readFile(new URL('native-prejoin-original-state.json',root),'utf8'));report.permission_restored_exact=JSON.stringify(report.restored.permission)===JSON.stringify(original.permission);assert.ok(report.permission_restored_exact&&report.restored.voice_connected&&report.restored.microphone_fgs);const room=(await read('/voice/participants')).channels.find(c=>c.channel_id===channelId);report.final={participants:room.participants.length,muted:room.participants.filter(p=>p.microphone_muted).length};assert.equal(report.final.participants,3);assert.equal(report.final.muted,report.before.muted)}catch(e){report.status='FAIL_RESTORATION';report.restoration_error_type=e.name}}
 await context.dispose();await writeFile(new URL('native-prejoin-result.json',root),JSON.stringify(report,null,2)+'\n');console.log(JSON.stringify(report))
}
