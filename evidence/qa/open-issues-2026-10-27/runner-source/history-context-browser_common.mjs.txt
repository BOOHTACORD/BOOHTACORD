  for(const key of ['at','before','after'])if(u.searchParams.has(key))entry[key+'_index']=d.ids.indexOf(u.searchParams.get(key))
  if(m[2]==='read-cursor'){const b=response.request().postDataJSON();entry.requested_index=d.ids.indexOf(b.message_id)}

  report.network.push(entry)

}
