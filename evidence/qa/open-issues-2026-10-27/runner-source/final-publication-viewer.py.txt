"""Append the closed already-open viewer observation without promoting its strict verdict."""
import sys, re
from pathlib import Path
Q = Path(__file__).resolve().parents[2]
sys.path.insert(0, str(Q / '.out/qa-archive-release'))
from common import *
src = Q / '.out/qa-phone-viewer589-master754b'
assert sha((src / 'summary.json').read_bytes()) == '824c7e5bc8d9cc10ec6ef02efade99b80581f7a4bad7e13003b9c6bf4f15c4a7'
files = read(src / 'publication-allowlist.json')['allow']
assert len(files) == 7
origins = []
for name in files:
    assert Path(name).name == name and name.endswith('.json')
    p = src / name
    raw = p.read_bytes()
    text = raw.decode('utf-8-sig')
    assert not any(n in text for n in needles)
    assert not re.search(r'[A-Za-z]:[/\\]Users[/\\]', text)
    dst = O / 'android-viewer589-master754b' / name
    assert not dst.exists()
    write(dst, read(p))
    origins.append({'source': name, 'private_original_SHA256': sha(raw),
                    'final_public_SHA256': sha(dst.read_bytes())})
write(O / 'android-viewer589-master754b/root-provenance.json', {
    'status': 'PASS_METADATA_ONLY', 'files': origins,
    'strict_verdict': 'NOT_PROVEN independent native history while modal remains open',
