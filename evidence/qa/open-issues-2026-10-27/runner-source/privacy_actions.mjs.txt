export async function messageActions(page,peer,prefix,message,report){
 report.message_action_checks={};const mark=s=>{report.message_action_stage=s}
 const controls=async r=>{await r.hover();await r.getByRole('button',{name:'Ответить',exact:true}).focus()}
 const row=page.locator('.message-row').filter({has:page.getByText(message.body,{exact:true})}).first()
 mark('reply');await controls(row);await row.getByRole('button',{name:'Ответить',exact:true}).click()
 mark('cancel_reply')
 const cancelReply=page.locator('.reply-target').getByRole('button');await expect(cancelReply).toBeVisible();await cancelReply.click();report.message_action_checks.reply_cancel='PASS'
 mark('edit');await controls(row);await row.getByRole('button',{name:'Изменить',exact:true}).click()
 const edited=message.body+' changed';await page.getByLabel('Изменённый текст сообщения').fill(edited)
 mark('edit_save');const patch=page.waitForResponse(r=>r.request().method()==='PATCH'&&new URL(r.url()).pathname.endsWith('/messages/'+message.id))
 await page.getByRole('button',{name:'Сохранить',exact:true}).click();assert.equal((await patch).status(),200)
 const next=page.locator('.message-row').filter({has:page.getByText(edited,{exact:true})}).first()
 mark('peer_edited_message');await expect(next).toBeVisible();await expect(peer.locator('.message-row').filter({has:peer.getByText(edited,{exact:true})})).toBeVisible();report.message_action_checks.edit_and_peer_update='PASS'
 await next.locator('.reaction-picker > summary').click()
 const reaction=next.locator('.reaction-palette').getByRole('button',{name:/^Реакция /}).first()
 mark('reaction');await reaction.click();await expect(reaction).toHaveAttribute('aria-pressed','true');await reaction.click();await expect(reaction).toHaveAttribute('aria-pressed','false');report.message_action_checks.reaction_toggle='PASS'
 mark('cancel_delete');await controls(next);page.once('dialog',d=>d.dismiss());await next.getByRole('button',{name:'Удалить',exact:true}).click()
 const history=await api(page,prefix+'/messages?at='+message.id+'&limit=20');assert.equal(history.status,200);assert.equal(history.body.messages.find(m=>m.id===message.id).deleted,false)
 const deletion=page.waitForResponse(r=>r.request().method()==='DELETE'&&new URL(r.url()).pathname.endsWith('/messages/'+message.id))
 mark('confirm_delete');await controls(next);page.once('dialog',d=>d.accept());await next.getByRole('button',{name:'Удалить',exact:true}).click();assert.equal((await deletion).status(),204)
 await expect(next).toHaveCount(0)
 return {reply_cancel:true,edit_http:200,peer_edit_visible:true,reaction_toggle:true,cancel_delete_preserved:true,confirm_delete_http:204}
}
