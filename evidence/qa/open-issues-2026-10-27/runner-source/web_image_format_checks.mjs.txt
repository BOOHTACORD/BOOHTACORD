import assert from 'node:assert/strict'
import {readFile} from 'node:fs/promises'
import {createHash} from 'node:crypto'
import {expect} from './privacy_fixture.mjs'
export const digest=b=>createHash('sha256').update(b).digest('hex')
export const format=b=>b.subarray(0,8).equals(Buffer.from([137,80,78,71,13,10,26,10]))?'PNG':b[0]===255&&b[1]===216?'JPEG':b.subarray(0,3).toString()==='GIF'?'GIF':'UNKNOWN'
export async function actualDownload(page,link,expected,paths){
 const wait=page.waitForEvent('download',{timeout:15000}).then(download=>({download}),error=>({error}))
 await link.click();const result=await wait;if(result.error)throw result.error
 const download=result.download;assert.equal(await download.failure(),null);const path=await download.path();paths.push(path)
 const bytes=await readFile(path),type=format(bytes),extension=download.suggestedFilename().split('.').at(-1).toLowerCase()
 return {actual_download_event:true,bytes:bytes.length,detected_format:type,suggested_extension:extension,
         matches_original_sha256:digest(bytes)===expected.sha256,extension_matches_encoding:({png:'PNG',jpg:'JPEG',jpeg:'JPEG',gif:'GIF'})[extension]===type}
}
export async function viewer(page,name,pixels,row,expected,paths){
 await page.bringToFront();const opener=page.getByRole('button',{name:'Открыть изображение '+name,exact:true});await expect(opener).toBeVisible();await opener.click()
 const dialog=page.getByRole('dialog',{name:'Просмотр изображения '+name,exact:true});await expect(dialog).toBeVisible()
 await expect.poll(()=>dialog.locator('img').evaluate((i,size)=>i.complete&&i.naturalWidth===size[0]&&i.naturalHeight===size[1],pixels)).toBe(true)
 row.actual_preview_decoded=true;row.viewer_rect=await dialog.evaluate(n=>{const r=n.getBoundingClientRect();return {x:r.x,y:r.y,width:r.width,height:r.height,viewport_width:innerWidth,viewport_height:innerHeight}})
 assert.ok(row.viewer_rect.x>=0&&row.viewer_rect.x+row.viewer_rect.width<=row.viewer_rect.viewport_width+1)
 const close=dialog.getByRole('button',{name:'Закрыть просмотр изображения',exact:true}),download=dialog.getByRole('link',{name:'Скачать',exact:true})
 row.modal_keyboard={initial_close_focused:await close.evaluate(n=>document.activeElement===n)}
 await page.keyboard.press('Tab');row.modal_keyboard.Tab_download_focused=await download.evaluate(n=>document.activeElement===n)
 row.modal_keyboard.Tab_focus_inside_dialog=await dialog.evaluate(n=>n.contains(document.activeElement))
 await page.keyboard.press('Shift+Tab');row.modal_keyboard.ShiftTab_close_focused=await close.evaluate(n=>document.activeElement===n)
 row.viewer_download=await actualDownload(page,download,expected,paths)
 await page.keyboard.press('Escape');await expect(dialog).toBeHidden();await expect(opener).toBeFocused();row.Escape_focus_return=true
}

 const rows=[];for(const suffix of['','/preview']){const r=await context.request.get(origin+'/api/v1'+path+suffix);rows.push(r.status())}return rows
}
