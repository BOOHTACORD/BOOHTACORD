class Client:

  try:
   c.request(method,'/api/v1'+path,body=raw,headers={'Origin':ORIGIN,'Cookie':self.cookie,'Content-Type':'application/json'})
   r=c.getresponse();raw=r.read();return r.status,json.loads(raw) if raw else None,dict(r.getheaders())
