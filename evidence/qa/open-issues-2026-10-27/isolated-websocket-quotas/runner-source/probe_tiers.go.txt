package main

import (
 "time"
)

func tiers(){
 for _,n:=range []int{1,10,100}{
  requireAll(batch(plan(0,n),origin));hold("tier_"+itoa(n),n,6);closeAll(n==1);snapshot("tier_"+itoa(n)+"_drained",0)
 }
 requireAll(batch(plan(0,490),origin));snapshot("deployment_before_race",490)
 before:=snapshot("deployment_race_baseline",490);candidates:=plan(490,30);results:=batch(candidates,origin);live:=accepted(results);require(len(live)==10,"deployment concurrent boundary count")
 for _,v:=range results{if v.socket==nil{denied(v,429)}}
 after:=snapshot("deployment_full",500);require(after["voice_platform_realtime_connections_rejected_total"]-before["voice_platform_realtime_connections_rejected_total"]==20,"deployment reject metric delta")
 var spare slot;found:=false
 for i,v:=range results{if v.socket==nil{spare=candidates[i];found=true;break}};require(found,"deployment spare absent")
 denied(dial(spare,origin),429);beforeRevalidation:=snapshot("full_before_revalidation",500);hold("tier_500",500,20);afterRevalidation:=snapshot("full_after_revalidation",500)
 record("full_tier_revalidation",map[string]any{"revalidation_call_delta":afterRevalidation["voice_platform_realtime_session_revalidation_seconds_count"]-beforeRevalidation["voice_platform_realtime_session_revalidation_seconds_count"],"revalidation_failures_delta":afterRevalidation["voice_platform_realtime_session_revalidation_failures_total"]-beforeRevalidation["voice_platform_realtime_session_revalidation_failures_total"],"note":"Actual API revalidation call metric, not inferred SQL query count"})
 closeSet(live[:1],false);snapshot("deployment_one_slot_released",499);requireAll(batch([]slot{spare},origin));snapshot("deployment_reconnect",500);closeAll(false);snapshot("deployment_drained",0)
 record("deployment_concurrent_quota",map[string]any{"prefill":490,"concurrent_boundary_attempts":30,"boundary_accepted":10,"boundary_rejected429":20,"independent_extra429":true,"full_active":500,"release499_reconnect500":true})
 for cycle:=1;cycle<=2;cycle++{requireAll(batch(plan(0,100),origin));hold("reconnect100_cycle_"+itoa(cycle),100,4);closeAll(false);snapshot("reconnect100_drained_"+itoa(cycle),0)}
 record("bounded_mass_reconnect",map[string]any{"connections_per_cycle":100,"cycles":2,"dial_concurrency_limit":20,"final_active":0})
}

func revokeOne(){
 s:=slot{103,0};live:=requireAll(batch([]slot{s},origin))[0];snapshot("revocation_before",1);started:=time.Now()
 status,_,_,e:=request(ctx,"POST","/api/v1/auth/logout",nil,owned.Accounts[s.Account].Cookies[s.Lane],-1);require(e==nil&&status==204,"own logout control")
 select{case <-live.done:case <-time.After(22*time.Second):panic("revoked socket did not close")}
 elapsed:=time.Since(started).Seconds();require(live.code.Load()==1008,"revoked socket policy close status");snapshot("revocation_drained",0)
 denied(dial(s,origin),401);requireAll(batch([]slot{{103,1}},origin));snapshot("sibling_session_still_usable",1);closeAll(false);snapshot("revocation_final_drained",0)
 record("actual_own_session_revocation",map[string]any{"logout_http":204,"close_code":1008,"elapsed_seconds":elapsed,"revoked_retry_http":401,"sibling_session_upgrades101":true,"slo_claim":"Observed bound only; no new general revalidation SLO claim"})
}
