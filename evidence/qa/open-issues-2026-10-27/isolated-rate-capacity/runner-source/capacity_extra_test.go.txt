package ratelimit
import("fmt";"sync";"sync/atomic";"testing";"time")
func TestConcurrent10000CapacityKeepsExhaustedBudget(t *testing.T){
 fixed:=time.Unix(2000,0);l,e:=New(Config{Limit:1,Window:time.Minute,MaxSources:10000,Now:func()time.Time{return fixed}});if e!=nil{t.Fatal(e)};l.Allow("protected")
 for i:=0;i<9984;i++{if _,ok:=l.Allow(fmt.Sprint("seed-",i));!ok{t.Fatal("prefill denied")}}
 var admitted,denied,protected atomic.Int64;var wg sync.WaitGroup;start:=make(chan struct{})
 for i:=0;i<32;i++{wg.Add(1);go func(n int){defer wg.Done();<-start;if _,ok:=l.Allow(fmt.Sprint("boundary-",n));ok{admitted.Add(1)}else{denied.Add(1)}}(i)}
 for i:=0;i<128;i++{wg.Add(1);go func(){defer wg.Done();<-start;if _,ok:=l.Allow("protected");ok{protected.Add(1)}}()};close(start);wg.Wait()
 if admitted.Load()!=15||denied.Load()!=17||protected.Load()!=0||l.Stats().ActiveSources!=10000||l.Stats().Saturated!=17{t.Fatalf("concurrent boundary wrong: admitted%d denied%d protected%d stats%+v",admitted.Load(),denied.Load(),protected.Load(),l.Stats())}
}
func TestExpired10000StorePrunesAtMost64AndDrains(t *testing.T){
 now:=time.Unix(3000,0);l,e:=New(Config{Limit:1,Window:time.Minute,MaxSources:10000,Now:func()time.Time{return now}});if e!=nil{t.Fatal(e)};for i:=0;i<10000;i++{l.Allow(fmt.Sprint("old-",i))};now=now.Add(time.Minute)
 previous:=10000;calls:=0;for i:=0;i<200;i++{if _,ok:=l.Allow(fmt.Sprint("fresh-",i));!ok{t.Fatal("expired capacity not released")};calls++;active:=l.Stats().ActiveSources;removed:=previous+1-active;if removed<0||removed>64||i==0&&removed!=64{t.Fatalf("cleanup removed%d",removed)};previous=active;if active==calls{break}}
 if calls!=157||l.Stats().ActiveSources!=157||len(l.expires)!=157{t.Fatalf("bounded cleanup did not drain: calls%d stats%+v heap%d",calls,l.Stats(),len(l.expires))}
}
func TestOriginalWindowBoundaryAtOneNanosecond(t *testing.T){
 now:=time.Unix(4000,0);l,_:=New(Config{Limit:1,Window:time.Minute,MaxSources:1,Now:func()time.Time{return now}});l.Allow("a");now=now.Add(time.Minute-time.Nanosecond)
 if retry,ok:=l.Allow("a");ok||retry!=time.Nanosecond{t.Fatalf("early reset %v %v",retry,ok)};now=now.Add(time.Nanosecond);if _,ok:=l.Allow("a");!ok{t.Fatal("exact boundary did not recover")}
}
func TestStaleHeapExpiryKeepsRenewedBudget(t *testing.T){
 now:=time.Unix(5000,0);l,_:=New(Config{Limit:1,Window:time.Minute,MaxSources:100,Now:func()time.Time{return now}});for i:=0;i<64;i++{l.Allow(fmt.Sprint("older-",i))};now=now.Add(time.Second);l.Allow("renewed");now=now.Add(time.Minute)
 if _,ok:=l.Allow("renewed");!ok{t.Fatal("renewal failed")};l.Allow("fresh");if _,ok:=l.Allow("renewed");ok{t.Fatal("stale heap node deleted renewed active entry")};if l.Stats().ActiveSources!=2{t.Fatalf("stats%+v",l.Stats())}
}
