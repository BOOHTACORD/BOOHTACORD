export async function call(page, target, method, ...args) {
  const result = await page.evaluate(async a => { try { return { ok: true, value: await window.qaPreview[a.target][a.method](...a.args) } } catch (e) { return { ok: false, errorType: e?.name ?? 'UnknownError' } } }, { target, method, args })
  if (!result.ok) throw Object.assign(new Error('QA_BROWSER_METHOD_FAILED'), { safeLabel: target + '.' + method, browserErrorType: result.errorType })
  return result.value
}

export async function until(label, fn, predicate, timeout = 18000) {
  const start = Date.now(); let value
  while (Date.now() - start < timeout) { value = await fn(); if (predicate(value)) return value; await new Promise(resolve => setTimeout(resolve, 100)) }
  throw Object.assign(new Error('QA_EXPECTATION_TIMEOUT'), { safeLabel: label, lastSafeValue: value })
}
let metricIndex = 0
export async function metrics() {
  const response = await fetch(cfg.metrics_url); if (!response.ok) throw new Error('QA_SFU_METRICS_HTTP')
  const raw = await response.text(); if (raw.length > 4194304) throw new Error('QA_SFU_METRICS_BOUND')
  await writeFile(path.join(P, 'private', 'metrics-' + (++metricIndex) + '.private.txt'), raw, 'utf8')
  const types = new Map(), sums = new Map()
  const allowed = new Set(['audio', 'video', 'screen_share', 'screen_share_audio', 'microphone', 'pub', 'sub', 'incoming', 'outgoing', 'success', 'failure', 'true', 'false', 'ok', 'error', 'subscribe', 'unsubscribe', 'published', 'unpublished'])
  for (const line of raw.split('\n')) {
    const type = /^# TYPE (\S+) (\S+)/.exec(line); if (type) { types.set(type[1], type[2]); continue }
    const match = /^(livekit_\S+?)(?:\{([^}]*)\})?\s+([-+\deE.]+)(?:\s.*)?$/.exec(line)
    if (!match || !/(track|participant_total|room_total)/.test(match[1]) || /_bucket$/.test(match[1])) continue
    const roleLabels = {}
    for (const pair of (match[2] ?? '').matchAll(/([a-z_]+)="([^"]*)"/g)) if (['kind', 'state', 'direction', 'type'].includes(pair[1]) && allowed.has(pair[2])) roleLabels[pair[1]] = pair[2]
    const key = match[1] + '|' + JSON.stringify(roleLabels); const previous = sums.get(key)
    sums.set(key, { name: match[1], type: types.get(match[1]) ?? 'unspecified', labels: roleLabels, value: (previous?.value ?? 0) + Number(match[3]) })
  }
  return { entries: [...sums.values()], raw_private_SHA256: sha(raw), raw_bytes: Buffer.byteLength(raw) }
}
export function activeVideo(metric) { const rows = metric.entries.filter(row => row.type === 'gauge' && /subscribed.*track|track.*subscribed/.test(row.name) && (!row.labels.kind || row.labels.kind === 'video')); return rows.length ? rows.reduce((sum, row) => sum + row.value, 0) : null }

  return sources
}
