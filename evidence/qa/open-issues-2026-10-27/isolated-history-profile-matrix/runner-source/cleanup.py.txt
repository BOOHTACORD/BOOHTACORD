    for row in data:row[5]=datetime.fromisoformat(row[5]).astimezone(timezone.utc).isoformat()
    out=io.StringIO(newline='');csv.writer(out,lineterminator='\n').writerows(data);digest=hashlib.sha256(out.getvalue().encode()).hexdigest();assert digest==d['seed_CSV_SHA256']
    r['dataset_preservation'].append({'dataset':d['alias'],'count':len(data),'seed_CSV_SHA256':digest,'server_data_matches_seed':True})

r['temporary_cleanup_FK_index_needed']=needs_index

    if needs_index:sql('CREATE INDEX '+temporary+' ON messages(reply_to_id);')
    sql('BEGIN;\nSET LOCAL statement_timeout=\'90s\';\nDELETE FROM channel_read_cursors WHERE channel_id IN ('+scopes+');\nDELETE FROM messages WHERE channel_id IN ('+scopes+');\nDELETE FROM channels WHERE category_id=\''+category+'\';\nDELETE FROM categories WHERE id=\''+category+"' AND name='qa_perf_"+c['owner_token']+"';\nCOMMIT;\n")

    if needs_index:sql('DROP INDEX IF EXISTS '+temporary+';')

r['message_index_definitions_restored_exactly']=True

values=sql('SELECT (SELECT count(*) FROM users),(SELECT count(*) FROM sessions),(SELECT count(*) FROM sessions WHERE revoked_at IS NULL),(SELECT count(*) FROM attachments),(SELECT count(*) FROM messages);\nSHOW server_version;').splitlines();counts=list(map(int,values[0].split('|')));expected=list(before['database_counts']);expected[1]+=1;assert counts==expected
r.update(database_counts_before=before['database_counts'],database_counts_after=counts,database_count_order=['users','all_sessions','unrevoked_rows_not_validity','attachments','messages'],PostgreSQL_version=values[1],own_profile_messages_removed=77750,own_control_sentinel_removed=1,own_channels_removed=22,own_category_removed=1,own_revoked_session_audit_rows_retained=1,users_or_attachments_created=0)

r['status']='PASS20_DATASET_HASHES_OWN_FIXTURE_REMOVAL_SESSION_REVOCATION_NINE_PRIVATE_SERVICES';r['finished_at_utc']=datetime.now(timezone.utc).isoformat();write(P/'environment-after.json',r)
