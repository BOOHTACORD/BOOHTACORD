import json,time
from urllib.parse import urlencode
from common import P,read,write,get,sha,now
QOE='sum(boohtacord_media_reports_total{platform="desktop_web",direction="receiver",state="playing"})'
NATIVE='sum(voice_platform_client_screen_reports_total{platform="desktop_web",direction="receiver",state="playing"})'
def prom(label):
    values={}
    for name,query in [('OTLP_QoE',QOE),('native_private',NATIVE)]:
        code,raw=get('prometheus','/api/v1/query?'+urlencode({'query':query}));assert code==200
        (P/'private'/('prom-'+label+'-'+name+'.json')).write_bytes(raw);d=json.loads(raw);assert d['status']=='success'
        xs=d['data']['result'];assert len(xs)<=1;values[name]={'series':len(xs),'value':float(xs[0]['value'][1]) if xs else None,'raw_SHA256':sha(raw)}
    return values
def walk(x):
    if isinstance(x,dict):
        if x.get('name')=='media.sample' and 'attributes' in x:yield x
        for v in x.values():yield from walk(v)
    elif isinstance(x,list):
        for v in x:yield from walk(v)
def attrs(s):return {x['key']:next(iter(x['value'].values())) for x in s['attributes']}
def after(c):

    while time.monotonic()-start<35:

        (P/'private'/('tempo-search-'+str(attempt)+'.json')).write_bytes(raw)
        for item in json.loads(raw).get('traces',[]):
            trace=item.get('traceID') or item.get('traceId');assert trace and all(x in '0123456789abcdef' for x in trace.lower())
            if trace not in traces:

        spans=[attrs(s) for raw in traces.values() for s in walk(json.loads(raw))]
        a=initial['OTLP_QoE']['value'];b=final['OTLP_QoE']['value']
        if a is not None and b is not None and b-a==1 and len(spans)==1:break
