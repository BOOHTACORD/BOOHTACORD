if not config.exists():

 hosts=[str(a) for a in network.hosts() if str(a)!=net['IPAddress']][32:220];assert len(hosts)>150
 values={'source_revision':SOURCE,'accounts':accounts,'wrong_password':secrets.token_urlsafe(30),'sources':{'nat':hosts[0:2],'account':hosts[10:35],'burst':hosts[50:114],'control':hosts[3],'churn':hosts[130]}}

if mode=='run':assert not (P/'run-result.json').exists() and not (P/'private/issued.private.json').exists()

env={**os.environ,'GOOS':'linux','GOARCH':'amd64','CGO_ENABLED':'0','GOTOOLCHAIN':'local'};build=subprocess.run(['go','build','-o',str(P/'login_probe'),*[str(f) for f in files]],cwd=Q/'backend',env=env,capture_output=True);(P/'build.private.log').write_bytes(build.stdout+build.stderr);assert build.returncode==0,'Inspect private compile log'
r={'at_utc':now(),'status':'RUNNING','mode':mode,'source_revision':SOURCE,'project':PROJECT,'production_touched':False,'binary_SHA256':hashlib.sha256((P/'login_probe').read_bytes()).hexdigest()};name='boohtacord-login-budget-probe'
argv=['docker','--context',CONTEXT,'run','--rm','--name',name,'--label','com.docker.compose.project='+PROJECT,'--label','qa.packet=login-budgets','--network','container:'+proxy['Id'],'--cpus','1','--memory','512m','--pids-limit','128','--read-only','--cap-drop','ALL','--security-opt','no-new-privileges','--mount','type=bind,source='+str(P/'login_probe')+',target=/qa-probe,readonly','--mount','type=bind,source='+str(P/'private')+',target=/work','--entrypoint','/qa-probe',api['Image'],mode]
done=threading.Event();db_samples=[]
def monitor():
 while not done.wait(4):
  try:
   values=docker('exec','-i',PROJECT+'-postgres-1','psql','-U','boohtacord_qa','-d','boohtacord_qa','-At',input=b"SELECT count(*),count(*) FILTER(WHERE state='active') FROM pg_stat_activity WHERE datname=current_database();\n");total,active=map(int,values.decode().strip().split('|'));db_samples.append({'at_utc':now(),'connections':total,'active_connections':active})

print(json.dumps(r));sys.exit(r.get('client_exit_code',2) if r['status']!='ERROR_CONTROLLER_OR_PROFILE' else 2)
