 const end=Date.now()+30000;
 while(Date.now()<end && !fs.existsSync(root+'qa648-transfer-sfu-ready'))await sleep(200);

 await route.continue();
});

console.log('QA648_NEW_DEVICE_STABLE');await marker('QA648_HANDOFF_STAYED_OUT',120000);
}finally{await browser.close()}})().catch(e=>{console.error('QA648_HANDOFF_CONTROL_FAILED',step,e.name);process.exitCode=1});
