import 'dart:async';
import 'dart:io';
import 'package:crypto/crypto.dart';
import 'package:flutter/material.dart';
import 'package:boohtacord_desktop/src/features/voice/roster_state/retry.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:integration_test/integration_test.dart';
import 'package:shared_preferences/shared_preferences.dart';
import 'package:flutter_secure_storage/flutter_secure_storage.dart';
import 'package:boohtacord_desktop/src/app.dart';
import 'package:boohtacord_desktop/src/app_state.dart';
import 'package:boohtacord_desktop/src/models.dart';
import 'package:boohtacord_desktop/src/services/api_client.dart';
import 'package:boohtacord_desktop/src/features/voice/admission/join.dart';

class LocalQaTls extends HttpOverrides {
  LocalQaTls(this.pin);
  final String pin;
  @override
  HttpClient createHttpClient(SecurityContext? context) =>
      super.createHttpClient(context)..badCertificateCallback = (cert, host, port) =>
          host == 'localhost' && port == 4910 && sha256.convert(cert.der).toString() == pin;
}

void main() {
  final binding = IntegrationTestWidgetsFlutterBinding.ensureInitialized();
  testWidgets('Windows returns to same voice after owned server outage', (tester) async {
    const origin=String.fromEnvironment('QA_API_BASE');
    const pin=String.fromEnvironment('QA_CERT_SHA256');
    const password=String.fromEnvironment('QA_PASSWORD');
    if (origin!='https://localhost:4910/api/v1' || pin.length!=64 || password.isEmpty) {
      throw StateError('Disposable QA endpoint, pinned certificate and credentials required');
    }
    HttpOverrides.global=LocalQaTls(pin);
    addTearDown(() => HttpOverrides.global=null);
    SharedPreferences.setMockInitialValues({});
    FlutterSecureStorage.setMockInitialValues({});
    final preferences=await SharedPreferences.getInstance();
    await preferences.setString('server_url',origin);
    final api=ApiClient();
    final state=AppState(api);
    addTearDown(state.dispose);
    await tester.pumpWidget(BoohtacordApp(state:state));
    await tester.runAsync(state.initialize);
    await tester.pumpAndSettle();
    if (state.phase!=AppPhase.ready) {
      final fields=find.byType(TextFormField);
      expect(fields,findsNWidgets(2));
      await tester.enterText(fields.at(0),'qa_admin');
      await tester.enterText(fields.at(1),password);
      FocusManager.instance.primaryFocus?.unfocus();
      await tester.pump();
      await tester.tap(find.text('Войти').last);
      for (var i=0;i<120 && state.phase!=AppPhase.ready;i++) {
        await tester.pump(const Duration(milliseconds:250));
      }
    }
    expect(state.phase,AppPhase.ready);
    await tester.runAsync(state.refreshTopology);
    expect(state.topology,isNotNull);
    await tester.runAsync(() async {
      if (!state.topology!.categories.any((c)=>c.name=='Voice QA')) {
        await api.createCategory('Voice QA');
        await state.refreshTopology();
      }
      final category=state.topology!.categories.firstWhere((c)=>c.name=='Voice QA');
      if (!category.channels.any((c)=>c.name=='Voice QA' && c.kind==ChannelKind.voice)) {
        await api.createChannel(categoryId:category.id,name:'Voice QA',kind:ChannelKind.voice);
        await state.refreshTopology();
      }
    });
    final channel=state.topology!.categories.expand((c)=>c.channels)
      .firstWhere((c)=>c.name=='Voice QA' && c.kind==ChannelKind.voice);
    await tester.runAsync(() => state.voice.joinVoice(channel,listenerOnly:true));
    if (state.voicePhase != VoicePhase.listener) {
      debugPrint('QA648_INITIAL_FAILURE_TYPE_${state.voice.lastVoiceAdmissionFailure.runtimeType}');
    }
    expect(state.voicePhase,VoicePhase.listener);
    expect(state.voiceChannel?.id,channel.id);
    final initialLease = state.voice.leaseId;
    expect(initialLease, isNotNull);
    await tester.pump();
    if (Platform.isAndroid) await binding.convertFlutterSurfaceToImage();
    await tester.pump();
    if (Platform.isAndroid) await binding.takeScreenshot('voice-before-outage');
    for (var i = 0; i < 10; i++) {
      await tester.runAsync(() => Future<void>.delayed(const Duration(seconds: 1)));
      await tester.pump();
      expect(state.voicePhase, VoicePhase.listener, reason: 'unstable pre-outage baseline');
      expect(state.voiceChannel?.id, channel.id);
    }
    debugPrint('ISSUE648_READY_FOR_OUTAGE');
    var sawRetryVisible=false;
    var sawInterruption=false;
    var recovered=false;
    var last=state.voicePhase;
    for (var n=0;n<1200;n++) {
      await tester.runAsync(() => Future<void>.delayed(const Duration(seconds:1)));
      await tester.pump();
      final phase=state.voicePhase;
      if (!sawRetryVisible && state.voice.voiceRecoveryActive && phase == VoicePhase.reconnecting) {
        expect(phase, VoicePhase.reconnecting);
        expect(state.voiceChannel?.id, channel.id);
        expect(find.textContaining('Восстанавливаем'), findsWidgets);
        if (Platform.isAndroid) await binding.takeScreenshot('voice-retry-visible');
        debugPrint('QA648_RETRY_VISIBLE');
        sawRetryVisible=true;
      }
      if (phase!=last) {debugPrint('ISSUE648_PHASE_${phase.name}');last=phase;}
      if (phase!=VoicePhase.listener && phase!=VoicePhase.connected) sawInterruption=true;
      if (sawInterruption && (phase==VoicePhase.listener || phase==VoicePhase.connected) &&
          state.voiceChannel?.id==channel.id) {recovered=true;break;}
    }
    await tester.pump();
    if (Platform.isAndroid) await binding.takeScreenshot('voice-after-outage');
    expect(sawInterruption,isTrue,reason:'No observed voice interruption');
    expect(recovered,isTrue,reason:'Did not return to original channel');
    expect(state.voice.leaseId, isNotNull);
    expect(state.voice.leaseId, isNot(initialLease), reason: 'Recovery reused the old lease');
    debugPrint('ISSUE648_FRESH_LEASE');
    expect(sawRetryVisible,isTrue,reason:'No visible recovery state observed during outage');
    debugPrint('ISSUE648_RECOVERED_SAME_CHANNEL');
    var rosterFresh = false;
    for (var i = 0; i < 60; i++) {
      await tester.runAsync(() => Future<void>.delayed(const Duration(seconds: 1)));
      await tester.pump();
      final roster = state.voiceRoster;
      if (roster.phase.name == 'fresh' && roster.watching &&
          (roster.voiceRosters ?? []).any((r) => r.channelId == channel.id && r.participants.length >= 2)) {
        rosterFresh = true;
        break;
      }
    }
    debugPrint(rosterFresh ? 'QA272_AUTO_FRESH' : 'QA272_AUTO_STUCK');
    final authoritative = await tester.runAsync(api.voiceParticipants);
    expect(authoritative!.any((r) => r.channelId == channel.id && r.participants.length >= 2), isTrue,
        reason: 'Owned API did not return the populated roster');
    if (!rosterFresh) {
      expect(state.voiceRoster.canRetry, isTrue);
      state.voiceRoster.retryVoiceRosters();
      for (var i = 0; i < 30 && state.voiceRoster.phase.name != 'fresh'; i++) {
        await tester.runAsync(() => Future<void>.delayed(const Duration(seconds: 1)));
        await tester.pump();
      }
      expect(state.voiceRoster.phase.name, 'fresh', reason: 'Manual recovery also failed');
      debugPrint('QA272_MANUAL_FRESH');
    }
    var peerVisible = false;
    for (var i = 0; i < 90; i++) {
      await tester.runAsync(() => Future<void>.delayed(const Duration(seconds: 1)));
      await tester.pump();
      if (find.text('qa_peer_check').evaluate().isNotEmpty) {
        peerVisible = true;
        break;
      }
    }
    debugPrint(peerVisible ? 'QA272_PEER_VISIBLE' : 'QA272_PEER_NOT_VISIBLE');
    if (Platform.isAndroid) {
      await tester.pump();
      await binding.takeScreenshot('roster-after-recovery');
    }
    debugPrint('QA272_DONE');
    expect(rosterFresh, isTrue, reason: 'Roster failed to recover automatically after server restart');
    expect(peerVisible, isTrue, reason: 'Other active participant absent from visible roster');
    await tester.runAsync(() => Future<void>.delayed(const Duration(seconds:20)));
  }, timeout: const Timeout(Duration(minutes:25)));
}
